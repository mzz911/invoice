function processFile(file,isScan,target){

function csvFor(list){
  const head='Tarikh Invoice,No Invoice,Supplier,Kategori,Cara Bayar,Split,Status,Tarikh Bayar,PV,Harga,Nota';
  const rows=list.map(x=>{
    const split = x.payMethod==='Split Payment' && x.split ? Object.entries(x.split).filter(([k,v])=>v>0).map(([k,v])=>`${k} ${v}`).join(' • ') : '';
    const esc=v=>`"${String(v||'').replace(/"/g,'""')}"`;
    return [x.date,x.inv,x.name,x.cat,x.payMethod,split,x.paid?'Dah Bayar':'Belum Bayar',x.payDate,x.pv,x.price,x.note].map(esc).join(',');
  });
  return head+'\n'+rows.join('\n');
}
