function nextPV(){ const y=String(new Date().getFullYear()).slice(2), m=String(new Date().getMonth()+1).padStart(2,'0'); const prefix=`PV-${y}${m}-`; let max=0; [...invoices, ...cancelledPVs].forEach(x=>{if(x.pv&&x.pv.startsWith(prefix)){max=Math.max(max, parseInt(x.pv.split('-')[2])||0)}}); return prefix+String(max+1).padStart(3,'0'); }
function filtered(){
 let list=[...invoices];
 if(tab==='unpaid') list=list.filter(x=>!x.paid);
 if(tab==='cancelled') return []; // handled separately
 if(query){ const q=query.toLowerCase(); list=list.filter(x=>[x.name,x.inv,x.pv,x.cat,x.jenis].join(' ').toLowerCase().includes(q)); }
 if(filters.supplier) list=list.filter(x=>x.name.toLowerCase().includes(filters.supplier.toLowerCase()));
 if(filters.status) list=list.filter(x=>filters.status==='paid'?x.paid:!x.paid);
 if(filters.pay) list=list.filter(x=>x.payMethod===filters.pay);
 if(filters.jenis) list=list.filter(x=>x.jenis===filters.jenis);
 // sort: paid by payDate desc then invDate, unpaid at bottom by invDate
 list.sort((a,b)=>{ if(a.paid&&b.paid) return (b.payDate||'').localeCompare(a.payDate||'')||(b.date||'').localeCompare(a.date||''); if(a.paid&&!b.paid) return -1; if(!a.paid&&b.paid) return 1; return (b.date||'').localeCompare(a.date||''); });
 return list;
}

     el.appendChild(d);
   }); return;
 }
 if(!list.length){$('#empty').style.display='block';$('#empty').textContent='Tiada invoice.';return}

 // group by payDate or invDate, then PV
 let groups={};
 list.forEach(x=>{ const key = x.paid? (x.payDate||x.date) : x.date; const gk = key+'|'+(x.paid? (x.pv||'') : 'BELUM'); if(!groups[gk]) groups[gk]=[]; groups[gk].push(x); });
 Object.entries(groups).forEach(([gk,items])=>{
   const [d,pv]=gk.split('|');
   const sum=items.reduce((s,x)=>s+Number(x.price||0),0);
   const g=document.createElement('div'); g.className='group';
   g.innerHTML=`<div><b>${new Date(d).toLocaleDateString('ms-MY',{day:'numeric',month:'short',year:'numeric'})}</b><div style="font-size:12px;color:#666">${items[0].paid?'Tarikh Bayar':'Tarikh Invoice'} • ${items.length} invoice ${pv&&pv!=='BELUM'?' • '+pv:''}</div></div><div style="text-align:right"><b>${fmt(sum)}</b><div style="font-size:12px;color:#666">Ketuk untuk tutup ▲</div></div>`;
   let collapsed=false; const wrap=document.createElement('div');
   items.forEach(x=>{
     const c=document.createElement('div'); c.className='card';
     const splitTxt = x.payMethod==='Split Payment' && x.split ? 'Split: '+Object.entries(x.split).filter(([k,v])=>v>0).map(([k,v])=>`${k.replace('Bank Transfer','Bank').replace('QR / E-Wallet','QR')} ${Number(v)}`).join(' • ') : '';
     c.innerHTML=`<input type="checkbox" ${selected.has(x.id)?'checked':''} style="width:26px;height:26px;margin-top:4px">
     <div class="thumb">${x.img?`<img src="${x.img}">`:'Resit'}</div>
     <div class="meta"><b>${x.name}</b><div>${x.date} • ${x.inv||'INV-'} • ${x.cat||x.jenis||''}</div>${splitTxt?`<div style="color:#111;font-weight:600">${splitTxt}</div>`:''}${x.payMethod&&x.payMethod!=='Split Payment'?`<span class="pill cash">${x.payMethod}</span>`:''}</div>
     <div class="right"><div class="price">${fmt(x.price)}</div><span class="pill ${x.paid?'paid':'unpaid'}">${x.paid?'Dah Bayar':'Belum Bayar'}</span>${x.paid&&x.pv?`<div><span class="pill pv">${x.pv}</span></div><div style="font-size:12px;color:#666;margin-top:4px">Bayar: ${x.payDate}</div>`:''}</div>`;

     wrap.appendChild(c);
   });
   g.onclick=()=>{collapsed=!collapsed; wrap.style.display=collapsed?'none':'block';};
   el.appendChild(g); el.appendChild(wrap);
 });
}
