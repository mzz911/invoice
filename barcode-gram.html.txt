<script>
function checkDigit(first12){
  let s=0;
  for(let i=0;i<12;i++){
    let d=parseInt(first12[i]);
    s += (i+1)%2==1 ? d*1 : d*3;
  }
  return (10 - (s%10))%10;
}
function buat(){
  let plu = document.getElementById('plu').value.padStart(5,'0');
  let nama = document.getElementById('nama').value;
  let hargaKg = parseFloat(document.getElementById('hargaKg').value);
  let gram = parseFloat(document.getElementById('gram').value);
  if(!gram){ alert('Masuk gram dulu!'); return; }
  let kg = gram/1000;
  let total = Math.round(kg * hargaKg * 100)/100;
  let totalCents = Math.round(total*100);
  let ppppp = String(totalCents).padStart(5,'0');
  let base12 = '02' + plu + ppppp;
  let c = checkDigit(base12);
  let full = base12 + c;

  document.getElementById('result').style.display='block';
  document.getElementById('namaOut').innerText = nama + ' - ' + plu;
  document.getElementById('beratOut').innerText = gram + 'g  (' + kg.toFixed(2) + 'kg)  x  RM' + hargaKg.toFixed(2) + '/kg';
  document.getElementById('hargaOut').innerText = 'RM' + total.toFixed(2);
  document.getElementById('numOut').innerText = full;
  JsBarcode('#barcode', full, {format:'EAN13', width:2, height:80, displayValue:false});
  // simpan last PLU & harga untuk ikan seterusnya
  localStorage.setItem('plu', plu);
  localStorage.setItem('hargaKg', hargaKg);
  localStorage.setItem('nama', nama);
}
document.getElementById('gram').addEventListener('keypress', function(e){ if(e.key==='Enter') buat(); });
</script>
